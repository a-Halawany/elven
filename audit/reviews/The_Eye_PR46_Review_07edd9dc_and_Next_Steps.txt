function query(rows, stats = {}, table='') {
  const filters = [], orders = []; let max = Infinity;
  const select = () => rows.filter(r => filters.every(f => f(r))).sort((a,b) => {
    for (const [k,d] of orders) { if(a[k]<b[k]) return d==='desc'?1:-1; if(a[k]>b[k]) return d==='desc'?-1:1; } return 0;
  }).slice(0,max);
  return { selectAll(){return this;}, select(){return this;}, $if(c,f){return c?f(this):this;},
    where(k,op,v){
      const f = { '=': r=>r[k]!=null && v!=null && String(r[k])===String(v), 'is': r=>r[k]===v, '<>': r=>r[k]!==v, '<': r=>r[k]<v, '>=': r=>r[k]>=v, '<=': r=>r[k]<=v, '>': r=>r[k]>v,
        'in': r=>v.includes(r[k]), 'not in': r=>!v.includes(r[k]) }[op];
      if(!f) throw Error('Unsupported predicate '+op); filters.push(f); return this;
    }, orderBy(k,d='asc'){orders.push([k,d]);return this;},limit(n){max=n;return this;},
    async execute(){stats.queries++; stats.trace.push(table); stats.timedTrace.push({table,at:NOW}); const out=select(); stats.readRows.push(...out.map(r=>({table,id:r.object_id??r.package_id??r.source_id??null}))); if(stats.onRead)stats.onRead(table,out); return out;},async executeTakeFirst(){return (await this.execute())[0];}
  };
}
